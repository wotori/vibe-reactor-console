#include <stdint.h>
#include <stdatomic.h>
#include "drivers/nvme.h"

struct nvme_queue {
    volatile uint32_t *doorbell;
    struct nvme_command *sq;
    struct nvme_completion *cq;
    uint16_t tail;
    uint16_t head;
    uint16_t depth;
    uint8_t phase;
};

/* Single producer owns each submission queue. */
static void submit(struct nvme_queue *q,
                    const struct nvme_command *cmd)
{
    uint16_t slot = q->tail;
    q->sq[slot] = *cmd;
    q->tail = (slot + 1) % q->depth;

    atomic_thread_fence(memory_order_release);
    mmio_write32(q->doorbell, q->tail);
}

int nvme_read(struct nvme_queue *q,
               uint64_t lba, uintptr_t buffer,
               uint16_t blocks)
{
    struct nvme_command cmd = {0};
    cmd.opcode = NVME_CMD_READ;
    cmd.nsid = 1;
    cmd.prp1 = virt_to_phys((void *)buffer);
    cmd.cdw10 = (uint32_t)lba;
    cmd.cdw11 = (uint32_t)(lba >> 32);
    cmd.cdw12 = blocks - 1;

    submit(q, &cmd);
    while ((mmio_read16(&q->cq[q->head].status)
            & 1u) != q->phase) {
        __asm__ volatile("pause");
    }

    uint16_t status = mmio_read16(
        &q->cq[q->head].status);
    if (++q->head == q->depth) {
        q->head = 0;
        q->phase ^= 1;
    }
    mmio_write32(q->doorbell + NVME_CQ_DB,
                  q->head);
    return (status >> 1) ? -1 : 0;
}
