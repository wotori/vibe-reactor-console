#version 450
layout(location = 0) out vec4 fragColor;
layout(binding = 0) uniform Scene {
    vec2 resolution;
    float time;
    float intensity;
} scene;

mat2 rotate(float angle) {
    float c = cos(angle), s = sin(angle);
    return mat2(c, -s, s, c);
}

float distanceField(vec3 p) {
    p.xz *= rotate(scene.time * 0.25);
    vec3 q = abs(p) - vec3(0.65);
    return length(max(q, 0.0))
         + min(max(q.x, max(q.y, q.z)), 0.0);
}

void main() {
    vec2 uv = (gl_FragCoord.xy * 2.0 - scene.resolution)
            / scene.resolution.y;
    vec3 origin = vec3(0.0, 0.0, -4.0);
    vec3 ray = normalize(vec3(uv, 1.8));
    float travel = 0.0;
    float glow = 0.0;

    for (int i = 0; i < 80; ++i) {
        vec3 p = origin + ray * travel;
        float d = distanceField(p);
        glow += 0.002 / (0.01 + abs(d));
        travel += max(abs(d), 0.01);
        if (travel > 12.0) break;
    }

    vec3 neon = vec3(0.05, 0.95, 0.72);
    fragColor = vec4(neon * glow * scene.intensity, 1.0);
}
