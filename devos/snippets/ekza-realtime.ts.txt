export type PublicRealtimeUserData = {
  nickname: string;
  avatar: string;
  avatarHeightScale?: number;
};

export type SetUserDataPayload = PublicRealtimeUserData & {
  /** Private random browser token used by the server to collapse duplicate tabs. */
  presenceToken?: string;
};

export type JoinSpacePayload =
  | string
  | { roomId: string; presenceToken?: string; claimId?: number };

export type LeaveSpacePayload =
  | string
  | { roomId: string; claimId?: number };

export type MovePayload = {
  position: Vector3Tuple;
  rotation: number;
  seq?: number;
  sentAt?: number;
};

export type MoveServerPayload = MovePayload &
  Partial<PublicRealtimeUserData> & {
    id: string;
    clientSeq?: number;
    serverSeq?: number;
    serverTime?: number;
  };
