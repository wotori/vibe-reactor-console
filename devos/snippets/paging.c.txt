#include <stdint.h>
#include <stddef.h>
#include "arch/x86_64/mmu.h"

#define PAGE_SHIFT  12u
#define PAGE_SIZE   (1ull << PAGE_SHIFT)
#define PTE_PRESENT (1ull << 0)
#define PTE_WRITE   (1ull << 1)
#define PTE_NX      (1ull << 63)
#define PHYS_MASK   0x000ffffffffff000ull

typedef uint64_t pte_t;
static pte_t kernel_pml4[512]
    __attribute__((aligned(4096)));

static inline void tlb_invalidate(uintptr_t va)
{
    __asm__ volatile("invlpg (%0)"
        : : "r"(va) : "memory");
}

static pte_t *walk_or_allocate(pte_t *table,
                               size_t index)
{
    if (!(table[index] & PTE_PRESENT)) {
        uintptr_t frame = frame_alloc_zeroed();
        table[index] = frame | PTE_PRESENT
                             | PTE_WRITE;
    }
    return phys_to_virt(table[index] & PHYS_MASK);
}

void map_page(uintptr_t va, uintptr_t pa,
              uint64_t flags)
{
    pte_t *pdpt = walk_or_allocate(kernel_pml4,
                                  (va >> 39) & 0x1ff);
    pte_t *pd = walk_or_allocate(pdpt,
                                (va >> 30) & 0x1ff);
    pte_t *pt = walk_or_allocate(pd,
                                (va >> 21) & 0x1ff);
    pt[(va >> PAGE_SHIFT) & 0x1ff] =
        (pa & PHYS_MASK) | flags | PTE_PRESENT;

    __asm__ volatile("mfence" : : : "memory");
    tlb_invalidate(va);
}

void paging_activate(void)
{
    uintptr_t root = virt_to_phys(kernel_pml4);
    __asm__ volatile("mov %0, %%cr3"
        : : "r"(root) : "memory");
    serial_write("[mmu] higher half online\n");
}
